# Персональна сторінка студента, система реєстрації та пошуку

UML діаграма сценарію роботи клієнтської частини

!theme bluegray

@startuml

left to right direction
actor "Студент" as Student
actor "Сервер" as Server

Student --> (Реєстрація на персональну сторінку)
Student --> (Авторизація)
Student --> (Оновлення персональних даних)
Student --> (Пошук інших студентів по імені)
Student --> (Пошук інших студентів по спеціальності)
Student --> (Переглянути дедлайни)

(Реєстрація на персональну сторінку) --> Server
(Авторизація) --> Server
(Оновлення персональних даних) --> Server
(Пошук інших студентів по імені) --> Server
(Пошук інших студентів по спеціальності) --> Server
(Переглянути дедлайни) --> Server

@enduml


UML діаграма сценарію роботи серверної частини

!theme bluegray
@startuml
left to right direction
actor "Сервер" as Server

package "Передача даних до клієнта" {
    (Інформація про авторизацію/реєстрацію)
    (Інформація про студента)
    (Інформація про дедлайни)
    (Інформація про пошук студентів)
}

package "Сервер" {
    (Обробка запиту)
    (Перевірка правильного\nприйнятого пакету)
    (Декодування команди \nприйнятої від клієнта)
}

package "База даних" {
    (Реєстрація на персональну сторінку)
	(Авторизація)
	(Оновлення персональних даних)
	(Пошук інших студентів по імені)
	(Пошук інших студентів по спеціальності)
	(Переглянути дедлайни)
}

Server --> (Інформація про авторизацію/реєстрацію)
Server --> (Інформація про студента)
Server --> (Інформація про дедлайни)
Server --> (Інформація про пошук студентів)
Server --> (Обробка запиту)
(Обробка запиту) --> (Декодування команди \nприйнятої від клієнта)
(Обробка запиту) --> (Перевірка правильного\nприйнятого пакету)
(Перевірка правильного\nприйнятого пакету) --> (Обробка запиту)

(Декодування команди \nприйнятої від клієнта) --> (Реєстрація на персональну сторінку)
(Декодування команди \nприйнятої від клієнта) --> (Авторизація)
(Декодування команди \nприйнятої від клієнта) --> (Оновлення персональних даних)
(Декодування команди \nприйнятої від клієнта) --> (Пошук інших студентів по імені)

(Декодування команди \nприйнятої від клієнта) --> (Переглянути дедлайни)
@enduml



Клієнтська частина процес авторизації користувача

!theme bluegray
@startuml
start
:Початок авторизації;
repeat:
:Введення пошти; <<input>>
:Введення паролю; <<input>>
:Запит на сервер; <<output>>
backward:Повідомлення про\nпомилку авторизації;
repeat while (Чи введені дані є у базі?) is (Ні) not (Так)
:Авторизація успішна;
stop
@enduml



Клієнтська частина процес реєстрації користувача

!theme bluegray
@startuml
start
:Початок реєстрації;
repeat:
:Введення ПІБ; <<input>>
:Введення пошти; <<input>>
:Введення паролю; <<input>>
:Введення cпеціальністі; <<input>>
:Введення групи; <<input>>
:Запит на сервер;<<output>>
backward:Повідомлення про\nпомилку реєстрації;
repeat while (Чи дані валідні?) is (Ні) not (Так)
stop
@enduml


Клієнтська частина оновлення персональних даних

!theme bluegray
@startuml
start

:Введення спеціальності;
:Введення групи;

:Запит на сервер;

if (Перевірка чи прийшла відповідь\nз сервера через певний проміжок часу) then (Так)
    :Повідомлення про успішне\nоновлення інформації про стужента;
else (Ні)
    :Повідомлення про збій або\nпро неіснування студента;
endif

stop
@enduml


Клієнтська частина сортування студентів

!theme bluegray
@startuml
start

:Введення умови сортування;
:Запит на сервер; <<output>>
if (Перевірка чи прийшла відповідь\nз сервера через певний проміжок часу) then (Так)
    :Виведення всіх студентів\nщо відповідають умові;
else (Ні)
    :Повідомлення про збій або\nпро неіснування студентів;
    ->;
endif

stop
@enduml



Клієнтська частина процес перегляду дедлайнів

!theme bluegray
@startuml
start
repeat :Запит на сервер для отримання спеціальностей; <<output>>
backward:Повідомлення про\nпомилку;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Вибір спеціальності;
repeat :Запит на сервер для отримання дедлайну для даної спеціальності; <<output>>
backward:Повідомлення про\nпомилку;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Віжображення дедлайнів;
stop
@enduml



Серверна частина для оновлення персональних даних


!theme bluegray
@startuml
start

:Отримання запиту про оновлення даних<

if (Чи дані валідні?) then (Так)
    fork
        :Запис спеціальності в БД;
    fork again
        :Запис групи в БД;
    end fork
else (Ні)
    :Повідомлення про збій;
endif

stop
@enduml


Серверна частина для перегляду дедлайнів певної спеціальності

!theme bluegray
@startuml
start

:Отримання даних; <<input>>

if (Чи дані валідні?) then (Так)
    if (Задана спеціальність існує?) then (Так)
	    :Запит до БД для отримання дедлайнів для заданої спеціальності;
		:Повернення дедлайнів;
	else (Ні)
	    :Помилка -- задана спеціальність не існує;
	endif
else (Ні)
    :Помилка -- дані не валідні;
endif

stop
@enduml



Серверна частина для пошуку студентів певної спеціальності

!theme bluegray
@startuml
start

:Отримання запиту про оновлення даних<

if (Спеціальність існує?) then (Так)
    :Запит до БД для отримання студентів для заданої спеціальності;
else (Ні)
    :Повідомлення про збій;
endif

stop
@enduml

Серверна частина для створення дедлайну

!theme bluegray
@startuml

start

:Отримання запиту про оновлення даних; <<input>>
:Перевірка отриманих данних\n на коректний тип даних та валідність;
if (Чи дані валідні?) then (Так)
    fork
        :Запис id предмету в БД;
    fork again
        :Запис початку дедлайну в БД;
    fork again
        :Запис кінця дедлайну в БД;
    end fork
else (Ні)
    :Повідомлення про некоректний тип даних;
endif


stop
@enduml
